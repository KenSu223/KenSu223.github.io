\section{Selected Publications \& Research}
\vspace{1pt}
\begin{enumerate}[leftmargin=1.3em, itemsep=3pt, topsep=0pt, parsep=0pt, label={[\arabic*]}]
  \item \textbf{T.~Su}\textsuperscript{*}, Y.~Hu\textsuperscript{*}, L.~Zhao, B.~Zhu, J.~Somasundaram, H.~Haque. ``Software World Models: From
  Consequence Prediction to Decision Value.'' \textit{arXiv:2610.04940}, 2026. \textit{(Under review; \textsuperscript{*}equal contribution.)}
  \begin{itemize}
    \item Formulated \textbf{system-level consequence prediction} for coding agents: predict the \textbf{blast set} --- the downstream services,
    libraries, and datastores a locally correct change will break --- \textit{before} execution, beyond what the agent itself observes.
    \item Built \textbf{Explore--Learn--Act}: execute counterfactual changes from restored system states, train Qwen2.5-Coder with \textbf{GRPO} on
    executed outcomes (plus an R-GCN graph-token variant with \textbf{SC-GRPO}), and act on per-consumer break probabilities.
    \item Blast-set F1 \textbf{0.431\,$\rightarrow$\,0.571} vs.\ static reachability and ranking regret \textbf{more than halved}; on held-out real libraries,
    agent return \textbf{0.453\,$\rightarrow$\,0.613}, above the CWM and Qwen-AgentWorld world-model baselines.
  \end{itemize}

  \item Y.~Hu, \textbf{T.~Su}, Y.~Sun, S.~Ma, Q.~Tan, L.~Zhao. ``Adaptive Memory Index Evolution for Agents.''
  \textit{arXiv:2610.34977}, 2026. \textit{(Under review.)}
  \begin{itemize}
    \item \textbf{AMIE}: a plug-in \textbf{memory index evolution} layer that distills an LLM teacher's keep/suppress judgments into a Qwen3-8B LoRA policy and
    accumulates them into persistent per-memory access state, with no change to stored content or host retraining. Improved downstream
    F1/accuracy in \textbf{all 15} comparisons (\textbf{5 memory systems} incl.\ Mem0, A-MEM $\times$ \textbf{3 benchmarks}) by up to \textbf{25\%} relative.
  \end{itemize}

  \item Y.~Hu\textsuperscript{*}, \textbf{T.~Su}\textsuperscript{*}, L.~Zhao, B.~Zhu, H.~Haque. ``LARGER: Lexically Anchored
  Repository Graph Exploration and Retrieval.'' \textit{arXiv:2605.16352}, 2026. \textit{(Under review; \textsuperscript{*}equal contribution.)}
  \begin{itemize}
    \item Anchored a coding agent's own lexical search hits into a repository graph and expanded confidence-filtered neighborhoods
    \textit{inside} its search loop (imports, call chains, code--test links) with no added agent actions; built \textbf{AST-based graph construction} end to end.
    \item \textbf{+13.9 Acc@5} and \textbf{+12.2 Recall@5} over the strongest of \textbf{7 baselines} (BM25, Agentless, LocAgent, SWE-agent, OpenHands,
    Claude Code, Codex) across \textbf{4 benchmarks} spanning localization, test generation, and codebase QA.
  \end{itemize}

  \item \textbf{T.~Su}, A.~Rafiei, M.~van Assen, R.~Sameni, G.~D.~Clifford, F.~Marzbanrad, N.~Katebi. ``Cross-Modal Generative
  Framework for Signal Translation from Fetal--Maternal Electrocardiograms to Fetal Doppler Waveforms.''
  \textit{IEEE EMBC 2026}. \textbf{Accepted --- Oral Presentation.}
  \begin{itemize}
    \item Designed a cross-modal generator (dilated convolutions + \textbf{cross-modal attention} + self-attention) for ECG-to-Doppler translation:
    PSD MSE \textbf{51\% below} baseline; ablations show naive concatenation adds nothing while cross-modal attention cuts error \textbf{39\%}.
  \end{itemize}

  \item \textbf{T.~Su}, N.~Barari, N.~Katebi. ``A Review of Multimodal Pretraining Strategies for Physiological Wearable Sensor
  Foundation Models.'' \textit{Under review}, 2026. --- Taxonomy of five multimodal pretraining paradigms over \textbf{22 papers} across PPG/ECG/EEG/ACC.
\end{enumerate}
