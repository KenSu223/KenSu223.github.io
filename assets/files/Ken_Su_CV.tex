\section{Selected Publications \& Research}
\vspace{1pt}
\begin{enumerate}[leftmargin=1.3em, itemsep=3pt, topsep=0pt, parsep=0pt, label={[\arabic*]}]
  \item Y.~Hu\textsuperscript{*}, \textbf{T.~Su}\textsuperscript{*}, L.~Zhao, B.~Zhu, H.~Haque. ``LARGER: Lexically Anchored
  Repository Graph Exploration and Retrieval.'' \textit{arXiv:2605.16352}, 2026. \textit{(Under review; \textsuperscript{*}equal contribution.)}
  \begin{itemize}
    \item Formalized \textbf{Lexically Anchored Structural Localization}: anchor a coding agent's own lexical search hits into a repository
    graph, then expand confidence-filtered neighborhoods \textit{inside} its existing search loop --- delivering structural evidence
    (imports, call chains, code--test links) with no external graph database, graph tools, or added agent actions.
    \item Built the system end to end: multi-language \textbf{AST-based graph construction}, edge weighting, and community detection for graph
    quality; sidecar storage, dynamic lexical propagation, and budget-aware filtering for retrieval efficiency.
    \item \textbf{+13.9 Acc@5} (tuned), \textbf{+11.8} (fixed hyperparameters), and \textbf{+12.2 Recall@5} over the strongest of \textbf{7 baselines}
    (BM25, Agentless, LocAgent, SWE-agent, OpenHands, Claude Code, Codex), with consistent gains across \textbf{4 benchmarks} spanning
    localization, test generation, and codebase QA.
  \end{itemize}

  \item \textbf{T.~Su}, A.~Rafiei, M.~van Assen, R.~Sameni, G.~D.~Clifford, F.~Marzbanrad, N.~Katebi. ``Cross-Modal Generative
  Framework for Signal Translation from Fetal--Maternal Electrocardiograms to Fetal Doppler Waveforms.''
  \textit{IEEE EMBC 2026}. \textbf{Accepted --- Oral Presentation.}
  \begin{itemize}
    \item Designed a cross-modal generative architecture (dilated convolutions + cross-modal attention + self-attention) synthesizing
    fetal Doppler envelopes from dual-lead ECG, trained on \textbf{885 synchronized segments from 39 pregnancies}: \textbf{PSD MSE 49.9\,dB\textsuperscript{2}}
    (\textbf{51\% below} the two-channel baseline) and \textbf{4.71\,bpm} heart-rate error.
    \item Showed via ablation that naive maternal-ECG concatenation yields no benefit while \textbf{cross-modal attention cuts PSD MSE 39\%} ---
    quantifying maternal--fetal cardiac coupling as selective and time-varying.
  \end{itemize}

  \item \textbf{T.~Su}, N.~Barari, N.~Katebi. ``A Review of Multimodal Pretraining Strategies for Physiological Wearable Sensor
  Foundation Models.'' \textit{Under review}, 2026. --- Synthesized \textbf{22 papers (2024--2026)} into a taxonomy of five pretraining
  paradigms (contrastive, generative, language-grounded, missing-modality, cross-modal distillation) across PPG/ECG/EEG/ACC.
\end{enumerate}
